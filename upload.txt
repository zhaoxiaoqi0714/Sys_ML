<script>
    document.addEventListener('DOMContentLoaded', function () {
        //////------- 是否有测试集交互按钮 -----------
        const hadTestCheckbox = document.getElementById('hadTestCheckbox');
        const testMatrixUpload = document.getElementById('testMatrixUpload');
        const testLabelUpload = document.getElementById('testLabelUpload');

        function toggleTestUpload() {
            if (hadTestCheckbox.checked) {
                testMatrixUpload.classList.remove('hidden');
                testLabelUpload.classList.remove('hidden');
            } else {
                testMatrixUpload.classList.add('hidden');
                testLabelUpload.classList.add('hidden');
            }
        }

        toggleTestUpload(); // 页面加载时初始化
        hadTestCheckbox.addEventListener('change', toggleTestUpload);

        //////------- 是否推荐 -----------
        const recommendCheckbox = document.getElementById('RecommendCheckbox');
        const recommendOptionsContainer = document.getElementById('recommendOptions');
        const customPreprocessingContainer = document.getElementById('customPreprocessingOptions');

        // 检查是否推荐的状态
        function checkRecommendState() {
            if (!recommendCheckbox.checked) {
                recommendOptionsContainer.style.display = 'block'; // 显示选项
            } else {
                recommendOptionsContainer.style.display = 'none'; // 隐藏选项
            }
        }

        // 检查自定义预处理模式是否被选中
        function checkCustomPreprocessingState() {
            const customRadio = document.querySelector('input[name="recommendOption"]:checked');
            if (customRadio && customRadio.value === 'custom') {
                customPreprocessingContainer.style.display = 'block'; // 显示自定义预处理选项
            } else {
                customPreprocessingContainer.style.display = 'none'; // 隐藏自定义预处理选项
            }
        }

        // 添加事件监听器
        recommendCheckbox.addEventListener('change', checkRecommendState);
        document.querySelectorAll('input[name="recommendOption"]').forEach(radio => {
            radio.addEventListener('change', checkCustomPreprocessingState);
        });

        // 页面加载时初始化
        checkRecommendState();
        checkCustomPreprocessingState();

        //////------- 是否集成 进行集成学习则必须可视化单机器学习结果 -----------
        const ensembleCheckbox = document.getElementById('ensembleCheckbox');
        const mlPlottingCheckbox = document.getElementById('mlPlottingCheckbox');

        ensembleCheckbox.addEventListener('change', function () {
            if (ensembleCheckbox.checked) {
                mlPlottingCheckbox.checked = true;
                mlPlottingCheckbox.disabled = true;
            } else {
                mlPlottingCheckbox.disabled = false;
            }
        });

        if (ensembleCheckbox.checked) {
            mlPlottingCheckbox.checked = true;
            mlPlottingCheckbox.disabled = true;
        }

        //////------- 生存分析中混杂因素输入控制-----------
        const inputField = document.getElementById('SA_cofactor_list');
        const addFactorButton = document.getElementById('addFactorButton');
        const errorMessage = document.getElementById('errorMessage');
        inputField.addEventListener('input', function () {
            const value = inputField.value;
            // 正则表达式：仅允许英文字符和分号
            if (!/^[a-zA-Z;]*$/.test(value)) {
                errorMessage.style.display = 'block'; // 显示错误提示
                inputField.value = value.replace(/[^a-zA-Z;]/g, ''); // 清除非英文字符
            } else {
                errorMessage.style.display = 'none'; // 隐藏错误提示
            }
        });

        addFactorButton.addEventListener('click', function () {
            const newFactor = prompt('请输入新的指标（仅支持英文）：');
            if (newFactor && /^[a-zA-Z]+$/.test(newFactor)) {
                const currentValue = inputField.value;
                if (currentValue) {
                    inputField.value = `${currentValue};${newFactor}`; // 添加分号分隔
                } else {
                    inputField.value = newFactor; // 直接添加
                }
            } else if (newFactor) {
                alert('仅支持英文字符！');
            }
        });

        //////------- 机器学习和生存分析只能选择一个，同时控制参数设置部分的显示与隐藏 -----------
        const mlAnalysisCheckbox = document.getElementById('mlAnalysisCheckbox');
        const survivalAnalysisCheckbox = document.getElementById('survivalAnalysisCheckbox');
        const mlSettings = document.getElementById('mlSettings');
        const saSettings = document.getElementById('saSettings');

        function updateSettings() {
            // 如果勾选了机器学习
            if (mlAnalysisCheckbox.checked) {
                survivalAnalysisCheckbox.checked = false; // 取消生存分析的勾选
                survivalAnalysisCheckbox.disabled = true; // 禁用生存分析
                mlSettings.style.display = 'block'; // 显示机器学习参数设置
                saSettings.style.display = 'none'; // 隐藏生存分析参数设置
            } else {
                survivalAnalysisCheckbox.disabled = false; // 启用生存分析
                mlSettings.style.display = 'none'; // 隐藏机器学习参数设置
            }

            // 如果勾选了生存分析
            if (survivalAnalysisCheckbox.checked) {
                mlAnalysisCheckbox.checked = false; // 取消机器学习的勾选
                mlAnalysisCheckbox.disabled = true; // 禁用机器学习
                saSettings.style.display = 'block'; // 显示生存分析参数设置
                mlSettings.style.display = 'none'; // 隐藏机器学习参数设置
            } else {
                mlAnalysisCheckbox.disabled = false; // 启用机器学习
                saSettings.style.display = 'none'; // 隐藏生存分析参数设置
            }
        }

        // 初始化状态
        updateSettings();
        mlAnalysisCheckbox.addEventListener('change', updateSettings);
        survivalAnalysisCheckbox.addEventListener('change', updateSettings);

        //////------- 获取表单元素 -----------
        document.getElementById('uploadForm').addEventListener('submit', (event) => {
            event.preventDefault(); // 阻止默认提交行为

            // 创建 FormData 对象
            const formData = new FormData(uploadForm);

            // 发送请求到后端
            fetch('upload_data/', { // 替换为你的后端接口地址
                method: 'POST',
                body: formData
            })
                .then(response => {
                    if (response.redirected) {
                        // 如果后端返回重定向响应，跳转到指定页面
                        window.location.href = response.url;
                    } else {
                        return response.json();
                    }
                })
                .then(data => {
                    if (data) {
                        alert(data.message); // 显示后端返回的消息
                    }
                })
                .catch(error => {
                    console.error('Error:', error);
                    alert('上传失败，请检查网络或联系管理员');
                });
        });
    });
</script>
